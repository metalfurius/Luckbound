            var store = new TrackingSaveStore(new FileSaveStore(saveDirectory, null));
            Func<ISaveStore> previousFactory = ApplicationBootstrapper.EditorSaveStoreFactoryForTests;
            ApplicationBootstrapper app = null;
            Keyboard keyboard = InputSystem.AddDevice<Keyboard>();
            Mouse mouse = InputSystem.AddDevice<Mouse>();
            Time.timeScale = 1f;
            try
            {
                yield return LoadBootstrappedApplication(store, value => app = value);
                Assert.That(app != null && app.SaveDataReady, Is.True);
                yield return LoadGuildHub();

                HubPreparationController preparation = UnityEngine.Object.FindAnyObjectByType<HubPreparationController>();
                Canvas canvas = FindHubCanvas();
                preparation.SelectClass(2);
                string selectedClass = preparation.SelectedClassId;
                string[] selectedSkills = preparation.SelectedSkillIds.ToArray();
                string profileBefore = JsonUtility.ToJson(app.Profile, false);
                int profileWritesBefore = store.ProfileWriteAttempts;
                int settingsWritesBefore = store.SettingsWriteAttempts;

                RectTransform boardRect = FindObject(canvas, "GuildIntroBoard").GetComponent<RectTransform>();
                Vector2 backgroundPoint = RectTransformUtility.WorldToScreenPoint(canvas.worldCamera,
                    boardRect.TransformPoint(new Vector3(0f, 46f, 0f)));

                Button firstCharacter = FindButton("GuildCharacterButton0");
                for (int frame = 0; frame < 3 && EventSystem.current.currentSelectedGameObject != firstCharacter.gameObject; frame++)
                    yield return null;
                Assert.That(EventSystem.current.currentSelectedGameObject, Is.SameAs(firstCharacter.gameObject),
                    "A real background click that clears EventSystem selection restores focus within the Guild view.");
                yield return PressUiKey(keyboard, Key.DownArrow);
                yield return PressUiKey(keyboard, Key.DownArrow);
                Assert.That(EventSystem.current.currentSelectedGameObject, Is.SameAs(FindButton("GuildCharacterButton2").gameObject));

                Button language = FindButton("GuildIntroConversationLanguageButton");
                EventSystem.current.SetSelectedGameObject(language.gameObject);
                store.FailNextSettingsWrite();

                string failedSettingsPayload = store.SettingsPayloads[settingsWritesBefore];
                Assert.That(FindText("GuildIntroText").text, Is.EqualTo(EnglishIntros[2]),

        [UnityTest]
        public IEnumerator E10GuildIntroRemainsReadableForAuthenticReadOnlyAndUnrelatedPendingProfiles()
        {
            string readOnlyDirectory = NewSaveDirectory("LuckboundE10GuildReadonly");
            string pendingDirectory = NewSaveDirectory("LuckboundE10GuildOtherPending");
            var readOnlyStore = new TrackingSaveStore(new FileSaveStore(readOnlyDirectory, null));
            var pendingStore = new TrackingSaveStore(new FileSaveStore(pendingDirectory, null));
            Func<ISaveStore> previousFactory = ApplicationBootstrapper.EditorSaveStoreFactoryForTests;
            ApplicationBootstrapper app = null;
            Time.timeScale = 1f;
            try
            {
                yield return LoadReadOnlyGuildProfile(readOnlyStore, value => app = value);
                Assert.That(app.ProfileLoadStatus, Is.EqualTo(SaveLoadStatus.LoadedPrimary));
                Assert.That(app.ProfileAccess, Is.EqualTo(SaveDataAccess.ReadOnlyLegacyEncounterPinMissing));
                Assert.That(app.SaveDataReady && !app.CanWriteProfile, Is.True);
                Assert.That(app.ActiveRun, Is.Null,
                    "The preserved legacy expedition is not required as an active runtime run to read guild text.");

                Task<SaveStorageSnapshot> readOnlyBeforeTask = readOnlyStore.ReadAsync(SaveDocument.Profile);
                yield return WaitForTask(readOnlyBeforeTask, "read readonly guild fixture before browsing");
                SaveStorageSnapshot readOnlyBefore = readOnlyBeforeTask.GetAwaiter().GetResult();
                string readOnlyProfileBefore = JsonUtility.ToJson(app.Profile, false);
                int readOnlyProgressBefore = app.ContractProgression.GetProgress(ContractCatalog.Ct01);
                bool readOnlyAcceptedBefore = app.ContractProgression.IsAccepted(ContractCatalog.Ct01);
                bool readOnlyClaimedBefore = app.ContractProgression.IsClaimed(ContractCatalog.Ct01);
                int readOnlyWritesBefore = readOnlyStore.ProfileWriteAttempts;

                yield return LoadGuildHub();
                SimulatePointerClick(FindButton("OpenGuildIntroButton"));
                SimulateSubmit(FindButton("GuildCharacterButton0"));
                Assert.That(FindText("GuildIntroText").text, Is.EqualTo(SpanishIntros[0]));
                SimulateSubmit(FindButton("GuildIntroBackButton"));
                SimulatePointerClick(FindButton("GuildIntroListCloseButton"));
                Assert.That(app.ActiveRun, Is.Null);
                Assert.That(app.ContractProgression.GetProgress(ContractCatalog.Ct01), Is.EqualTo(readOnlyProgressBefore));
                Assert.That(app.ContractProgression.IsAccepted(ContractCatalog.Ct01), Is.EqualTo(readOnlyAcceptedBefore));
                Assert.That(app.ContractProgression.IsClaimed(ContractCatalog.Ct01), Is.EqualTo(readOnlyClaimedBefore));
                Assert.That(JsonUtility.ToJson(app.Profile, false), Is.EqualTo(readOnlyProfileBefore));
                Assert.That(readOnlyStore.ProfileWriteAttempts, Is.EqualTo(readOnlyWritesBefore));
                Task<SaveStorageSnapshot> readOnlyAfterTask = readOnlyStore.ReadAsync(SaveDocument.Profile);
                yield return WaitForTask(readOnlyAfterTask, "read readonly guild fixture after browsing");
                SaveStorageSnapshot readOnlyAfter = readOnlyAfterTask.GetAwaiter().GetResult();
                Assert.That(readOnlyAfter.PrimaryJson, Is.EqualTo(readOnlyBefore.PrimaryJson));
                Assert.That(readOnlyAfter.BackupJson, Is.EqualTo(readOnlyBefore.BackupJson));

                yield return LoadBootstrappedApplication(pendingStore, value => app = value);
                Assert.That(app != null && app.SaveDataReady && app.CanWriteProfile, Is.True);
                pendingStore.FailNextProfileWrite();
                ExpeditionRunConfiguration run = ExpeditionRunConfigurationFactory.Create("class.bellator",
                    ExpeditionRunConfiguration.HumanRaceId,
                    new[] { "skill.bellator.circular_cut", "skill.bellator.guard" });
                Task<SaveWriteResult> beginTask = app.BeginExpeditionAsync(run);
                yield return WaitForTask(beginTask, "create the unrelated pending expedition write fixture");
                SaveWriteResult beginResult = beginTask.GetAwaiter().GetResult();
                Assert.That(beginResult.Succeeded, Is.False);
                Assert.That(app.HasPendingExpeditionWrite, Is.True);
                Assert.That(app.HasPendingContractProgression, Is.False);
                Assert.That(app.ActiveRun, Is.Null);
                Assert.That(app.HasActiveExpedition, Is.False);

                Task<SaveStorageSnapshot> pendingBeforeTask = pendingStore.ReadAsync(SaveDocument.Profile);
                yield return WaitForTask(pendingBeforeTask, "read unrelated pending fixture before guild browsing");
                SaveStorageSnapshot pendingBefore = pendingBeforeTask.GetAwaiter().GetResult();
                string pendingProfileBefore = JsonUtility.ToJson(app.Profile, false);
                string pendingPayloadBefore = app.PendingProfileExport;
                int pendingProgressBefore = app.ContractProgression.GetProgress(ContractCatalog.Ct01);
                int pendingWritesBefore = pendingStore.ProfileWriteAttempts;

                yield return LoadGuildHub();
                SimulatePointerClick(FindButton("OpenGuildIntroButton"));
                SimulateSubmit(FindButton("GuildCharacterButton4"));
                Assert.That(FindText("GuildIntroText").text, Is.EqualTo(SpanishIntros[4]));
                SimulatePointerClick(FindButton("GuildIntroConversationCloseButton"));

                Assert.That(app.HasPendingExpeditionWrite, Is.True);
                Assert.That(app.PendingProfileExport, Is.EqualTo(pendingPayloadBefore));
                Assert.That(app.HasPendingContractProgression, Is.False);
                Assert.That(app.ActiveRun, Is.Null);
                Assert.That(app.HasActiveExpedition, Is.False);
                Assert.That(app.ContractProgression.GetProgress(ContractCatalog.Ct01), Is.EqualTo(pendingProgressBefore));
                Assert.That(JsonUtility.ToJson(app.Profile, false), Is.EqualTo(pendingProfileBefore));
                Assert.That(pendingStore.ProfileWriteAttempts, Is.EqualTo(pendingWritesBefore));
                Task<SaveStorageSnapshot> pendingAfterTask = pendingStore.ReadAsync(SaveDocument.Profile);
                yield return WaitForTask(pendingAfterTask, "read unrelated pending fixture after guild browsing");
                SaveStorageSnapshot pendingAfter = pendingAfterTask.GetAwaiter().GetResult();
                Assert.That(pendingAfter.PrimaryJson, Is.EqualTo(pendingBefore.PrimaryJson));
                Assert.That(pendingAfter.BackupJson, Is.EqualTo(pendingBefore.BackupJson));
            }
            finally
            {
                ApplicationBootstrapper.EditorSaveStoreFactoryForTests = previousFactory;
                app = ApplicationBootstrapper.Instance;
                if (app != null) UnityEngine.Object.Destroy(app.gameObject);
                Time.timeScale = 1f;
                DeleteSaveDirectory(readOnlyDirectory);
                DeleteSaveDirectory(pendingDirectory);
            }
        }

        private static IEnumerator LoadReadOnlyGuildProfile(
            TrackingSaveStore store, Action<ApplicationBootstrapper> ready)
        {
            ApplicationBootstrapper app = null;
            yield return LoadBootstrappedApplication(store, value => app = value);
            Assert.That(app != null && app.CanWriteProfile, Is.True,
                "The readonly V6 fixture starts writable and reaches readonly status through a real legacy reload.");

            Task<ContractProgressionCommitResult> accepted = app.AcceptContractAsync(ContractCatalog.Ct01);
            yield return WaitForTask(accepted, "accept CT01 for the readonly V6 fixture");
            Assert.That(accepted.GetAwaiter().GetResult().Persisted, Is.True);
            for (int index = 0; index < 3; index++)
            {
                Task<ContractProgressionCommitResult> progress = app.RecordContractProgressEventAsync(
                    ContractProgressEvent.CombatRoomCompleted, "test/e10-guild/readonly-room-" + index);
                yield return WaitForTask(progress, "record readonly V6 fixture progress " + index);
                Assert.That(progress.GetAwaiter().GetResult().Persisted, Is.True);
            }

            ExpeditionRunConfiguration run = ExpeditionRunConfigurationFactory.Create("class.bellator",
                ExpeditionRunConfiguration.HumanRaceId,
                new[] { "skill.bellator.circular_cut", "skill.bellator.guard" });
            Task<SaveWriteResult> begin = app.BeginExpeditionAsync(run);
            yield return WaitForTask(begin, "save the legacy active expedition for the readonly V6 fixture");
            Assert.That(begin.GetAwaiter().GetResult().Succeeded, Is.True);
            Assert.That(app.HasActiveExpedition, Is.True);

            UnityEngine.Object.Destroy(app.gameObject);
            yield return null;
            SceneManager.LoadScene("Bootstrap", LoadSceneMode.Single);
            for (int frame = 0; frame < 120 &&
                (ApplicationBootstrapper.Instance == null || !ApplicationBootstrapper.Instance.SaveDataReady ||
                 SceneManager.GetActiveScene().name != "Menu"); frame++) yield return null;
            app = ApplicationBootstrapper.Instance;
            Assert.That(app != null && app.SaveDataReady, Is.True);
            Assert.That(app.ProfileLoadStatus, Is.EqualTo(SaveLoadStatus.LoadedPrimary));
            Assert.That(app.ProfileAccess, Is.EqualTo(SaveDataAccess.ReadOnlyLegacyEncounterPinMissing));
            Assert.That(app.CanWriteProfile, Is.False);
            Assert.That(app.ActiveRun, Is.Null);
            Assert.That(app.ContractProgression.IsComplete(ContractCatalog.Ct01), Is.True);
            ready(app);
        }

        private static IEnumerator LoadBootstrappedApplication(ISaveStore store, Action<ApplicationBootstrapper> ready)
        {
            ApplicationBootstrapper old = ApplicationBootstrapper.Instance;
            if (old != null) UnityEngine.Object.Destroy(old.gameObject);
            yield return null;
            ApplicationBootstrapper.EditorSaveStoreFactoryForTests = () => store;
            SceneManager.LoadScene("Bootstrap", LoadSceneMode.Single);
            for (int frame = 0; frame < 120 &&
                (ApplicationBootstrapper.Instance == null || !ApplicationBootstrapper.Instance.SaveDataReady); frame++)
                yield return null;
            ready(ApplicationBootstrapper.Instance);
        }

        private static IEnumerator LoadGuildHub()
        {
            SceneManager.LoadScene("Hub", LoadSceneMode.Single);
            HubPreparationController preparation = null;
            for (int frame = 0; frame < 60 && preparation == null; frame++)
            {
                preparation = UnityEngine.Object.FindAnyObjectByType<HubPreparationController>();
                if (preparation == null) yield return null;
            }
            Assert.That(preparation, Is.Not.Null);
            for (int frame = 0; frame < 2; frame++) yield return null;
        }

        private static IEnumerator WaitForTask<T>(Task<T> task, string operation)
        {
            Assert.That(task, Is.Not.Null, operation + " returned no task.");
            for (int frame = 0; frame < 300 && !task.IsCompleted; frame++) yield return null;
            Assert.That(task.IsCompleted, Is.True, operation + " did not complete within 300 frames.");
        }

        private static void AssertGuildTextLayout(Text text, bool wrapped = false)
        {
            Assert.That(text, Is.Not.Null);
            Canvas.ForceUpdateCanvases();
            RectTransform rect = text.rectTransform;
            Vector2 extents = rect.rect.size;
            TextGenerationSettings settings = text.GetGenerationSettings(extents);
            var preferredSizeGenerator = new TextGenerator();
            preferredSizeGenerator.Populate(text.text, settings);
            Assert.That(text.pixelsPerUnit, Is.GreaterThan(0f),
                text.name + " must expose a positive pixels-per-unit scale.");
            float preferredWidth = preferredSizeGenerator.GetPreferredWidth(text.text, settings) / text.pixelsPerUnit;
            float preferredHeight = preferredSizeGenerator.GetPreferredHeight(text.text, settings) / text.pixelsPerUnit;
            Assert.That(text.fontSize * (320f / 480f), Is.GreaterThanOrEqualTo(14f),
                text.name + " must render at or above 14 px at 320x180.");
            if (!wrapped)
            {
                Assert.That(preferredWidth, Is.LessThanOrEqualTo(extents.x + 0.5f),
                    text.name + " preferred width must fit its available rect.");
            }
            Assert.That(preferredHeight, Is.LessThanOrEqualTo(extents.y + 0.5f),
                text.name + " preferred height must fit its available rect.");
            var visibilityGenerator = new TextGenerator();
            visibilityGenerator.Populate(text.text, settings);
            Assert.That(visibilityGenerator.characterCountVisible, Is.EqualTo(text.text.Length),
                text.name + " must display every authored character with its configured overflow mode.");
        }

        private static IEnumerator SimulateScreenPointerClick(Mouse mouse, Vector2 screenPosition)
        {
            InputSystem.QueueStateEvent(mouse, new MouseState { position = screenPosition });
            InputSystem.Update();
            yield return null;
            InputSystem.QueueStateEvent(mouse,
                new MouseState { position = screenPosition }.WithButton(MouseButton.Left));
            InputSystem.Update();
            yield return null;
            InputSystem.QueueStateEvent(mouse, new MouseState { position = screenPosition });
            InputSystem.Update();
            for (int frame = 0; frame < 2; frame++) yield return null;
        }

        private static IEnumerator PressUiKey(Keyboard keyboard, Key key)
        {
            InputSystem.QueueStateEvent(keyboard, new KeyboardState(key));
            InputSystem.Update();
            for (int frame = 0; frame < 2; frame++) yield return null;
            InputSystem.QueueStateEvent(keyboard, new KeyboardState());
            InputSystem.Update();
            yield return null;
        }

        private static string NewSaveDirectory(string prefix)
        {
            string directory = Path.Combine(Path.GetTempPath(), prefix, Guid.NewGuid().ToString("N"));
            Directory.CreateDirectory(directory);
            return directory;
        }

        private static void DeleteSaveDirectory(string directory)
        {
            string tempRoot = Path.GetFullPath(Path.GetTempPath()).TrimEnd(Path.DirectorySeparatorChar) + Path.DirectorySeparatorChar;
            string fullDirectory = Path.GetFullPath(directory);
            if (!fullDirectory.StartsWith(tempRoot, StringComparison.OrdinalIgnoreCase))
                throw new InvalidOperationException("Refusing to delete a save directory outside the system temp root.");
            if (Directory.Exists(fullDirectory)) Directory.Delete(fullDirectory, recursive: true);
        }

        private sealed class TrackingSaveStore : ISaveStore
        {
            private readonly FileSaveStore _inner;
            private bool _failNextProfileWrite;
            private bool _failNextSettingsWrite;

            public TrackingSaveStore(FileSaveStore inner) { _inner = inner; }

            public int ProfileWriteAttempts { get; private set; }
            public int SettingsWriteAttempts { get; private set; }
            public System.Collections.Generic.List<string> SettingsPayloads { get; } = new System.Collections.Generic.List<string>();
            public string GetPrimaryPath(SaveDocument document) => _inner.GetPrimaryPath(document);
            public Task<SaveStorageSnapshot> ReadAsync(SaveDocument document) => _inner.ReadAsync(document);
            public void FailNextProfileWrite() => _failNextProfileWrite = true;
            public void FailNextSettingsWrite() => _failNextSettingsWrite = true;

            public async Task WriteAsync(SaveDocument document, string json, string lastKnownGoodPrimaryJson)
            {
                if (document == SaveDocument.Profile)
                {
                    ProfileWriteAttempts++;
                    if (_failNextProfileWrite)
                    {
                        _failNextProfileWrite = false;
                        throw new IOException("Injected profile write failure for the unrelated pending-write fixture.");
                    }
                }
                else
                {
                    SettingsWriteAttempts++;
                    SettingsPayloads.Add(json);
                    if (_failNextSettingsWrite)
                    {
                        _failNextSettingsWrite = false;
                        throw new IOException("Injected settings write failure for the language retry fixture.");
                    }
                }
                await _inner.WriteAsync(document, json, lastKnownGoodPrimaryJson);
            }
        }
    }
